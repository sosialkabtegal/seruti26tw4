// GANTI DENGAN URL WEB APP DARI DEPLOYMENT APPS SCRIPT ANDA const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbzKwRKunO-GCrSRjTK_iPzqzPTa6on9i1YCCGhGAjl8koD_7qgFTHk7MVSEfUqZo7cd/exec'; const form = document.querySelector('#formEntri'); const btnSubmit = form.querySelector('button[type="submit"]') || form.querySelector('button'); // Fungsi bantu untuk membaca file input menjadi Base64 function fileToBase64(file) { return new Promise((resolve, reject) => { if (!file) return resolve(null); const reader = new FileReader(); reader.onload = () => resolve(reader.result); reader.onerror = error => reject(error); reader.readAsDataURL(file); }); } form.addEventListener('submit', async function (e) { e.preventDefault(); // Ubah status tombol saat loading const teksAsliTombol = btnSubmit.innerHTML; btnSubmit.disabled = true; btnSubmit.innerHTML = 'Sedang mengirim data...'; try { const formData = new FormData(form); const payload = {}; // 1. Ambil nilai teks biasa for (let [key, value] of formData.entries()) { if (typeof value === 'string') { payload[key] = value; } } // 2. Baca file Foto 1 jika ada const inputFoto1 = form.querySelector('input[name="foto1"]'); if (inputFoto1 && inputFoto1.files[0]) { const file1 = inputFoto1.files[0]; payload['foto1_base64'] = await fileToBase64(file1); payload['foto1_name'] = file1.name; payload['foto1_type'] = file1.type; } // 3. Baca file Foto 2 jika ada const inputFoto2 = form.querySelector('input[name="foto2"]'); if (inputFoto2 && inputFoto2.files[0]) { const file2 = inputFoto2.files[0]; payload['foto2_base64'] = await fileToBase64(file2); payload['foto2_name'] = file2.name; payload['foto2_type'] = file2.type; } // 4. Kirim data ke Apps Script menggunakan URLSearchParams agar tidak terjadi error CORS const params = new URLSearchParams(); for (const key in payload) { params.append(key, payload[key]); } const response = await fetch(SCRIPT_URL, { method: 'POST', body: params }); const result = await response.json(); if (result.status === 'success') { alert('Data berhasil disimpan ke Spreadsheet!'); form.reset(); } else { alert('Terjadi kesalahan: ' + result.message); } } catch (error) { console.error('Error saat submit:', error); alert('Gagal mengirim data. Pastikan koneksi internet stabil dan URL Web App sudah benar.'); } finally { btnSubmit.disabled = false; btnSubmit.innerHTML = teksAsliTombol; } });
